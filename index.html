Powers of two

Return a new array where each element is 2 raised to the power of the original. [1,2,3] → [2,4,8]. Solved three ways. 

// for
const result = [];
for (let i = 0; i < numbers.length; i++) {
  result.push(Math.pow(2, numbers[i]));
}

// forEach
numbers.forEach(function (number) {
  result.push(Math.pow(2, number));
});

// map
return numbers.map(function (number) {
  return Math.pow(2, number);
});

Even / odd / N/A

map each number to "even" or "odd". Anything that is not a number becomes "N/A". 

return values.map(function (value) {
  if (typeof value !== "number") return "N/A";

  // NaN is technically of type "number", so we check it separately
  if (Number.isNaN(value)) return "N/A";

  if (value % 2 === 0) return "even";

  return "odd";
});

FizzBuzz

÷3 → Fizz, ÷5 → Buzz, ÷ both → Fizz Buzz, otherwise the number itself. 

return numbers.map(function (number) {
  // the "both" case must be first, or 15 would stop at "Fizz"
  if (number % 3 === 0 && number % 5 === 0) return "Fizz Buzz";

  if (number % 3 === 0) return "Fizz";

  if (number % 5 === 0) return "Buzz";

  return number;
});
